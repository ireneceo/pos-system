### 1단계: DB 통계 집계

```bash
cd /var/www/dev-backend && node -e "
require('dotenv').config();
const Content = require('./models/Content');
const { Op } = require('sequelize');
(async () => {
  const now = Date.now();
  const threeMonthsAgo = new Date(now - 90*24*60*60*1000);

  // Top 10 by views
  const top = await Content.findAll({
    where: { type: 'blog', status: 'published' },
    order: [['view_count', 'DESC']],
    limit: 10,
    attributes: ['id', 'language', 'title', 'slug', 'view_count', 'published_at', 'content_tier', 'target_persona']
  });
  console.log('=== TOP 10 (by views) ===');
  top.forEach(p => console.log(p.view_count.toString().padStart(6), '|', p.language, '|', p.content_tier || '?', '|', p.target_persona || '?', '|', p.title.substring(0,60)));

  // Rewrite candidates: published > 90 days ago, view_count < 50
  const underperformers = await Content.findAll({
    where: {
      type: 'blog', status: 'published',
      published_at: { [Op.lt]: threeMonthsAgo },
      view_count: { [Op.lt]: 50 }
    },
    order: [['view_count', 'ASC']],
    limit: 20,
    attributes: ['id', 'language', 'title', 'slug', 'view_count', 'published_at']
  });
  console.log('\\n=== REWRITE CANDIDATES (>90d old, <50 views) ===');
  underperformers.forEach(p => console.log(p.view_count.toString().padStart(6), '|', p.language, '|', p.title.substring(0,60)));

  // Publishing cadence (last 4 weeks)
  const fourWeeksAgo = new Date(now - 28*24*60*60*1000);
  const recent = await Content.findAll({
    where: {
      type: 'blog', status: 'published',
      published_at: { [Op.gte]: fourWeeksAgo }
    },
    attributes: ['language', 'content_tier', 'published_at']
  });
  console.log('\\n=== Last 4 weeks cadence ===');
  const byLang = {};
  recent.forEach(r => { byLang[r.language] = (byLang[r.language]||0) + 1; });
  Object.entries(byLang).forEach(([l,n]) => console.log('  ', l, ':', n, 'posts'));
  const byTier = {};
  recent.forEach(r => { byTier[r.content_tier||'uncategorized'] = (byTier[r.content_tier||'uncategorized']||0) + 1; });
  console.log('  by tier:');
  Object.entries(byTier).forEach(([t,n]) => console.log('  ', t, ':', n));

  // Coverage gaps: blogs without translation group (single-language only)
  const orphans = await Content.findAll({
    where: { type: 'blog', status: 'published', translation_group_id: null },
    attributes: ['id', 'language', 'title']
  });
  console.log('\\n=== TRANSLATION GAPS (no translation_group_id) ===');
  console.log('Orphan posts:', orphans.length);

  // Group analysis: which groups are missing translations?
  const grouped = await Content.findAll({
    where: { type: 'blog', status: 'published', translation_group_id: { [Op.ne]: null } },
    attributes: ['translation_group_id', 'language', 'title']
  });
  const byGroup = {};
  grouped.forEach(g => {
    byGroup[g.translation_group_id] = byGroup[g.translation_group_id] || { langs: [], titles: {} };
    byGroup[g.translation_group_id].langs.push(g.language);
    byGroup[g.translation_group_id].titles[g.language] = g.title;
  });
  const incomplete = Object.entries(byGroup).filter(([,v]) => v.langs.length < 3);
  console.log('\\nIncomplete translation groups (need EN+MS+ZH, missing at least 1):');
  incomplete.forEach(([gid, v]) => {
    const missing = ['en','ms','zh'].filter(l => !v.langs.includes(l));
    console.log('  group', gid, '| has:', v.langs.join(','), '| missing:', missing.join(','), '| title:', Object.values(v.titles)[0].substring(0,50));
  });

  process.exit(0);
})();
"
```

### 2단계: 분석 리포트 작성

Claude가 출력을 받아 자연어 리포트:

```
## 📊 블로그 성과 감사 (YYYY-MM-DD)

### 🏆 Top 3 성공작
1. "..." - 2,341 views, 3개월간 유입 리더
2. ...

### ⚠️ 리라이트 우선 후보 (5편)
- 기준: 3개월 이상 지났으나 50 views 미만
- 공통점: [Claude가 파악한 패턴]
- 권장 조치: [제목 리프레시 / CTA 추가 / 길이 보강 / 삭제]

### 📅 발행 리듬 (최근 4주)
- 목표: 주 3편 × 3언어 = 12 row/주
- 실제: X row/주
- 부족 영역: [Tactical 부족 / MS 번역 누락 / etc]

### 🌏 번역 갭
- Orphan 글 N개 (translation_group 없음)
- 3언어 미완성 그룹 N개 → 우선 번역 추천 Top 5

### 🎯 다음 주 액션 아이템
1. ...
2. ...
```

### 3단계: 자동 다음 액션 제안

- 리라이트 후보 Top 3 → `/블로그-초안 rewrite=<id>` 실행 권장
- 번역 갭 → `/블로그-초안 translate=<group_id> langs=ms,zh` 실행 권장
- 발행 부족 → `/블로그-리서치` 로 주제 뱅크 보충 권장
