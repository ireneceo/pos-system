# /블로그-발행 — 블로그 초안을 발행 상태로 전환

**호출**: `/블로그-발행 <translation_group_id>` 또는 `/블로그-발행 <content_id>` (단일 언어만 발행)

---

## Claude가 수행할 작업

### 1단계: 대상 컨텐츠 식별

```bash
cd /var/www/dev-backend && node -e "
require('dotenv').config();
const Content = require('./models/Content');
(async () => {
  const arg = <group_id or content_id>;
  // If numeric and looks like group: find all rows with that translation_group_id
  const rows = await Content.findAll({ where: { translation_group_id: arg } });
  if (!rows.length) {
    // fallback: find by id
    const one = await Content.findByPk(arg);
    if (one) console.log('[single]', one.id, one.language, one.title, one.status);
  } else {
    rows.forEach(r => console.log('[group=' + arg + ']', r.id, r.language, r.title, r.status));
  }
  process.exit(0);
})();
"
```

### 2단계: status='published' + published_at 설정

```bash
cd /var/www/dev-backend && node -e "
require('dotenv').config();
const Content = require('./models/Content');
(async () => {
  const rows = await Content.findAll({ where: { translation_group_id: <id>, status: 'draft' } });
  for (const r of rows) {
    await r.update({ status: 'published', published_at: r.published_at || new Date() });
    console.log('Published:', r.language, '/', r.slug);
  }
  process.exit(0);
})();
"
```

### 3단계: sitemap 갱신 (향후 추가)

별도 sitemap-contents.xml 생성 로직이 있으면 트리거. 현재는 WordPress 스타일 sitemap.xml 자동 갱신 아직 미구현 — Phase 5에서 추가 예정.

### 4단계: 검증

```bash
for lang in en ms zh; do
  curl -s "https://dev.purplehere.com/api/contents/public/blog/<slug>?lang=$lang" | node -e "
    let s=''; process.stdin.on('data',d=>s+=d); process.stdin.on('end',()=>{
      const d = JSON.parse(s);
      console.log(process.argv[1], ':', d.post?.status, '|', d.post?.title);
    });" "$lang"
done
```
